    <button class="reset" onclick="resetGame()">Reset Game</button>
  </div>

  <script>
    const DAILY_LIMIT = 500;

    let points = 0;
    let currentQuestion;

    const questions = [
      {
        question: "What is 5 + 5?",
        answers: ["8", "10", "15", "20"],
        correct: "10"
      },
      {
        question: "What color do you get by mixing blue and yellow?",
        answers: ["Purple", "Orange", "Green", "Red"],
        correct: "Green"
      },
      {
        question: "How many days are in a week?",
        answers: ["5", "6", "7", "8"],
        correct: "7"
      },
      {
        question: "What planet do we live on?",
        answers: ["Mars", "Venus", "Earth", "Jupiter"],
        correct: "Earth"
      },
      {
        question: "What is 12 × 2?",
        answers: ["20", "22", "24", "26"],
        correct: "24"
      }
    ];

    function newQuestion() {
      if (points >= DAILY_LIMIT) {
        document.getElementById("question").innerHTML =
          '<span class="complete">🎉 You reached 500 points!</span>';

        document.getElementById("answers").innerHTML = "";
        return;
      }

      currentQuestion =
        questions[Math.floor(Math.random() * questions.length)];

      document.getElementById("question").textContent =
        currentQuestion.question;

      const answersDiv = document.getElementById("answers");
      answersDiv.innerHTML = "";

      currentQuestion.answers.forEach(answer => {
        const button = document.createElement("button");

        button.textContent = answer;
        button.className = "answer";

        button.onclick = () => checkAnswer(answer);

        answersDiv.appendChild(button);
      });
    }

    function earn(amount) {
      const remaining = DAILY_LIMIT - points;

      if (remaining <= 0) {
        return;
      }

      const earned = Math.min(amount, remaining);

      points += earned;

      updateDisplay();
    }

    function checkAnswer(answer) {
      const message = document.getElementById("message");

      if (answer === currentQuestion.correct) {
        earn(50);
        message.textContent = "✅ Correct! +50 points";
        message.style.color = "#86efac";
      } else {
        message.textContent = "❌ Incorrect! Try the next question.";
        message.style.color = "#fca5a5";
      }

      setTimeout(() => {
        message.textContent = "";
        newQuestion();
      }, 700);
    }

    function updateDisplay() {
      document.getElementById("points").textContent = points;

      const percentage = (points / DAILY_LIMIT) * 100;

      document.getElementById("progress").style.width =
        percentage + "%";
    }

    function resetGame() {
      points = 0;
      updateDisplay();
      document.getElementById("message").textContent = "";
      newQuestion();
    }

    // Start the game
    updateDisplay();
    newQuestion();
  </script>
